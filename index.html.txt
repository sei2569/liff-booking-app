<script>
    // 1. ใส่ LIFF ID ที่ได้จาก LINE Developers
    const LIFF_ID = "ใส่_LIFF_ID_ของคุณที่นี่";
    // 2. ใส่ Web App URL ที่ได้จาก Google Apps Script
    const GAS_URL = "ใส่_WEB_APP_URL_จาก_GAS_ที่นี่";

    // เริ่มต้นทำงาน LIFF
    async function initializeLiff() {
        try {
            await liff.init({ liffId: LIFF_ID });
            // ถ้าอยากดึงชื่อ LINE อัตโนมัติ สามารถใช้ liff.getProfile() ตรงนี้ได้
        } catch (err) {
            console.error("LIFF Initialization failed", err);
        }
    }
    initializeLiff();

    // จัดการเมื่อกดปุ่ม Submit ฟอร์ม
    document.getElementById('bookingForm').addEventListener('submit', function(e) {
        e.preventDefault(); // ป้องกันเว็บโหลดใหม่
        
        const submitBtn = document.getElementById('submitBtn');
        const loader = document.getElementById('loader');
        
        // ล็อกปุ่มและโชว์สถานะโหลด
        submitBtn.disabled = true;
        loader.style.display = 'block';

        // ดึงข้อมูลจากฟอร์ม
        const payload = {
            date: document.getElementById('date').value,
            time: document.getElementById('time').value,
            name: document.getElementById('name').value,
            empId: document.getElementById('empId').value,
            room: document.getElementById('room').value
        };

        // ส่งข้อมูลไปที่ Google Apps Script
        fetch(GAS_URL, {
            method: 'POST',
            body: JSON.stringify(payload)
        })
        .then(response => response.json())
        .then(data => {
            if(data.status === 'success') {
                alert('จองห้องประชุมสำเร็จ!');
                // ปิดหน้าจอ LIFF ทันทีที่จองเสร็จ
                if (liff.isInClient()) {
                    liff.closeWindow();
                } else {
                    window.location.reload();
                }
            } else {
                alert('เกิดข้อผิดพลาด: ' + data.message);
                submitBtn.disabled = false;
                loader.style.display = 'none';
            }
        })
        .catch(error => {
            console.error('Error:', error);
            alert('ไม่สามารถเชื่อมต่อระบบได้');
            submitBtn.disabled = false;
            loader.style.display = 'none';
        });
    });
</script>
